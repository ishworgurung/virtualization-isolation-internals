1//! Replay the ABA interleaving against an untagged and a tagged Treiber stack.
2//!
3//! There is only ONE real thread here. "T1" is the `pop_with_hook` call that
4//! reads `head`, then pauses; "T2" is the closure it runs in that pause. This
5//! reproduces, deterministically, an interleaving that with real threads would
6//! happen only rarely (T1 would have to be descheduled at exactly that point).
7//!
8//! Nodes are numbered 0, 1, 2 and shown as A, B, C. The step-by-step picture
9//! of what happens is in the module docs of `src/aba.rs`.
10//!
11//! Run:  cargo run --release --bin aba_demo
12use lockfree::aba::{Pop, TreiberStack, NIL};
13
14/// Node index -> letter (0 -> 'A', 1 -> 'B', ...) for readable output.
15/// `b'A'` is the ASCII code of 'A' as a byte (65). Letters are consecutive in
16/// ASCII, so 65 + 1 = 66 = 'B', 65 + 2 = 67 = 'C'; `as char` turns the number
17/// back into a character.
18fn name(i: u32) -> char {
19    (b'A' + i as u8) as char
20}
21/// Render a stack snapshot top-down as "A -> B -> C".
22/// Read the chain left to right: iterate the indices, turn each into a
23/// one-letter `String`, collect them into a `Vec`, and join with " -> ".
24/// (`|&i|` destructures the `&u32` the iterator yields into a plain `u32`.)
25fn show(v: &[u32]) -> String {
26    v.iter().map(|&i| name(i).to_string()).collect::<Vec<_>>().join(" -> ")
27}
28
29/// Run the scenario once, with or without the generation tag.
30fn run(tagged: bool) {
31    const A: u32 = 0;
32    const B: u32 = 1;
33    const C: u32 = 2;
34
35    println!("--- {} head ---", if tagged { "TAGGED (index + generation)" } else { "UNTAGGED (index only)" });
36    let treiber_stack = TreiberStack::new(3, tagged);
37    treiber_stack.push(C);
38    treiber_stack.push(B);
39    treiber_stack.push(A);
40    println!("initial stack            : {}", show(&treiber_stack.snapshot()));
41
42    // Thread 1 starts pop(): reads head=A and A.next=B, then is descheduled.
43    let outcome = treiber_stack.pop_with_hook(|| {
44        println!("T1 read head=A, next=B ... T1 is now paused before its CAS");
45        // Thread 2 runs to completion in the gap.
46        let a = treiber_stack.pop().unwrap();
47        let b = treiber_stack.pop().unwrap();
48        println!("T2 popped {} then {}", name(a), name(b));
49        treiber_stack.scribble_next(b, NIL); // T2 keeps using B for something else
50        println!("T2 is now using {} elsewhere (its next pointer changed)", name(b));
51        treiber_stack.push(a);
52        println!("T2 pushed {} back        : stack is {}", name(a), show(&treiber_stack.snapshot()));
53        println!("T1 resumes and executes CAS(head, A -> B)");
54    });
55
56    match outcome {
57        // Untagged: head is A again, so the CAS "A? then B" succeeds wrongly.
58        Pop::Popped(i) => println!(
59            "T1's CAS SUCCEEDED and returned {}.\n  stack now              : {}\n  EXPECTED               : (A already re-pushed) A -> C\n  => CORRUPTION: node C is lost, B is both 'in use by T2' and on the stack.",
60            name(i),
61            show(&treiber_stack.snapshot())
62        ),
63        // Tagged: head is A with a newer generation, so the CAS fails.
64        Pop::Retry => println!(
65            "T1's CAS FAILED (generation changed), so T1 would retry.\n  stack still            : {}\n  => correct.",
66            show(&treiber_stack.snapshot())
67        ),
68        Pop::Empty => unreachable!(), // the stack held three nodes
69    }
70    println!();
71}
72
73fn main() {
74    run(false);
75    run(true);
76}
