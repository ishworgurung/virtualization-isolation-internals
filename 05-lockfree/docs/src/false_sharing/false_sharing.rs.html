1//! False sharing, measured.
2//!
3//! Two threads each hammer THEIR OWN counter. Logically there is zero sharing.
4//! Case 1 places the counters adjacent (same 64-byte cache line); case 2 pads
5//! them onto separate lines. Same work, same instructions, different speed.
6//!
7//! ## What is a cache line?
8//!
9//! The unit of memory a CPU moves between RAM and its caches: **64 bytes** on
10//! x86-64 and most ARM (128 on Apple M-series). Check yours with
11//! `getconf LEVEL1_DCACHE_LINESIZE`.
12//!
13//! RAM is slow (~100 ns per access), so each core keeps copies of recently used
14//! memory in small, fast caches (L1 ~1 ns, then L2, L3). Caches never hold
15//! individual variables, only whole 64-byte blocks. The line is the unit of
16//! *transfer*, not the size of your data: reading a 1-byte `u8` still fetches
17//! the whole 64-byte block of memory it sits in, i.e. your byte plus its 63
18//! neighbouring bytes (other variables, array elements, struct fields), on the
19//! bet that nearby data will be needed soon:
20//!
21//! ```text
22//!   line at 0x1000 (64 bytes):
23//!   [ neighbours ... | your u8 at 0x1008 | ... more neighbours ]
24//! ```
25//!
26//! So a line holds whatever happens to be in that 64-byte block, and any
27//! two variables in the same block share a line.
28//!
29//! Which line an address lives on: divide by 64 (same as `addr >> 6`):
30//!
31//! ```text
32//!   0x1000..0x103F            -> line 0x40
33//!   0x1040..0x107F            -> line 0x41
34//!   a at 0x1008, b at 0x1010  -> both on line 0x40
35//! ```
36//!
37//! That is what the `line` closure in `main` computes.
38//!
39//! ## Why case 1 is slow
40//!
41//! A core may write a line only while it holds the **only** copy (cache
42//! coherence, the MESI protocol). Before writing, it invalidates every other
43//! core's copy. With `a` and `b` (8 bytes each) on the same line:
44//!
45//! 1. Core 0 increments `a` -> the line is invalidated in core 1's cache.
46//! 2. Core 1 increments `b` -> it must fetch the line back (tens of ns),
47//!    invalidating core 0's copy.
48//! 3. Repeat ITERS times.
49//!
50//! Each thread only touches its own counter, but the hardware tracks ownership
51//! per *line*, not per variable, so the line ping-pongs between the cores.
52//! That is **false sharing**: nothing is logically shared, yet the cores behave
53//! as if it were. `CachePadded` gives each counter its own 64-byte line, so each
54//! core keeps its line in L1 and never waits for the other (~4x faster on the
55//! machine in the README). More background and diagrams: `src/padded.rs`.
56//!
57//! Run:  cargo run --release --bin false_sharing
58use lockfree::padded::CachePadded;
59use std::sync::atomic::{AtomicU64, Ordering::Relaxed};
60use std::time::Instant;
61
62const ITERS: u64 = 5_000_000;
63
64/// Memory layout (addresses illustrative):
65///
66/// ```text
67///   Adjacent:  0x1000 [ a: 8 bytes ][ b: 8 bytes ] ...       one 64-byte line
68///   Padded:    0x1000 [ a: 8 bytes ][ 56 bytes padding ]      line 0x40
69///              0x1040 [ b: 8 bytes ][ 56 bytes padding ]      line 0x41
70/// ```
71///
72/// Both counters sit next to each other (16 bytes total), so they almost
73/// always share a cache line. Not guaranteed — the struct is only 8-byte
74/// aligned and could straddle a line boundary — which is why `main` prints it.
75///
76/// Note these are already `AtomicU64`s and still false-share: an atomic is
77/// 8 bytes, the same as a `u64`, so atomicity does nothing to separate
78/// variables onto different lines. Only padding (`Padded` below) does. See
79/// "Why not just use `AtomicU64` everywhere" in `src/padded.rs`.
80struct Adjacent {
81    a: AtomicU64,
82    b: AtomicU64,
83}
84
85/// Each counter owns its cache line.
86struct Padded {
87    a: CachePadded<AtomicU64>,
88    b: CachePadded<AtomicU64>,
89}
90
91/// Increment `x` ITERS times.
92///
93/// `x: &AtomicU64` is a shared reference: several threads may hold one at
94/// once. That is fine because atomics are designed to be modified through a
95/// shared `&` (a plain `u64` could not be). `for _ in 0..ITERS` repeats ITERS
96/// times; `_` means "I don't need the loop counter".
97fn hammer(x: &AtomicU64) {
98    for _ in 0..ITERS {
99        // Relaxed fetch_add: no ordering needed, we only want raw write traffic
100        // on that line. (On x86 it compiles to a `lock`-prefixed add; being
101        // atomic, the compiler may not merge the ITERS increments into one.)
102        x.fetch_add(1, Relaxed);
103    }
104}

106/// Run `f` and return how long it took, in milliseconds.
107///
108/// `<F: FnOnce()>` makes this a GENERIC function: `F` stands for "whatever
109/// type the caller passes", with the constraint that it must be callable once
110/// with no arguments (every closure has its own unique type, so we can't name
111/// it; generics let us accept any of them). `-> f64` is the return type.
112/// `Instant::now()` reads a monotonic clock; `elapsed()` is the time since
113/// then; `* 1e3` converts seconds to milliseconds. The last expression without
114/// a `;` is the function's return value (no `return` keyword needed).
115fn time<F: FnOnce()>(f: F) -> f64 {
116    let t0 = Instant::now();
117    f();
118    t0.elapsed().as_secs_f64() * 1e3
119}
120
121fn main() {
122    let (aa, bb) = (AtomicU64::new(0), AtomicU64::new(0));
123    let (pa, pb) = (AtomicU64::new(0), AtomicU64::new(0));
124    let adj = Adjacent {a: aa, b: bb};
125    let pad = Padded { a: CachePadded(pa), b: CachePadded(pb) };
126
127    // ---------------------------------------------------------------------
128    // Step 1: show WHERE the counters live in memory.
129    // ---------------------------------------------------------------------
130    //
131    // `line` is a CLOSURE: a small unnamed function stored in a variable.
132    //   |p: *const AtomicU64|   <- parameter list between the pipes, with type
133    //   p as usize / 64         <- body; its value is the return value
134    // Called like a normal function: `line(x)`.
135    //
136    // `*const AtomicU64` is a RAW POINTER: a plain memory address, like a C
137    // pointer. We never read through it (that would need `unsafe`); we only
138    // want the number. `p as usize` reinterprets the address as an integer
139    // (`as` is Rust's explicit cast; Rust never converts types silently).
140    //
141    // Dividing by 64 gives the cache-line number: integer division drops the
142    // remainder, i.e. the byte's offset within its line (same as `>> 6`, since
143    // 64 = 2^6). E.g. 0x1008 / 64 = 0x40 and 0x1010 / 64 = 0x40 -> same line.
144    //
145    // `&adj.a` makes a REFERENCE (`&AtomicU64`, a safe, borrow-checked
146    // pointer). Rust automatically converts a reference into the raw pointer
147    // `line` asks for.
148    let line = |p: *const AtomicU64| (p as usize) >> 6;
149
150    // `println!` is a MACRO (the `!` marks macros): it checks the format string
151    // at compile time. Each `{}` is filled by the next argument in order;
152    // `{:#x}` means "hexadecimal with a 0x prefix". The third value is a
153    // `bool` from `==`, printed as `true`/`false`.
154    println!("adjacent: a and b on cache lines {:#x} / {:#x}  (same line: {})",
155        line(&adj.a), line(&adj.b), line(&adj.a) == line(&adj.b));
156
157    // `&*pad.a`, read right to left:
158    //   pad.a     the CachePadded<AtomicU64> wrapper
159    //   *pad.a    `*` dereferences; our `Deref` impl (src/padded.rs) makes it
160    //             yield the AtomicU64 inside
161    //   &*pad.a   `&` takes a reference to that inner AtomicU64
162    // Plain `&pad.a` would be a `&CachePadded<AtomicU64>` (same address here,
163    // since the counter sits at the start of the wrapper): the wrong type for
164    // `line`, and Rust won't auto-unwrap when converting to a raw pointer.
165    // The trailing `\n` in the string prints an extra blank line.
166    println!("padded  : a and b on cache lines {:#x} / {:#x}  (same line: {})\n",
167        line(&*pad.a), line(&*pad.b), line(&*pad.a) == line(&*pad.b));
168
169    // ---------------------------------------------------------------------
170    // Step 2: a helper that runs a measurement 3 times and keeps the fastest.
171    // ---------------------------------------------------------------------
172    //
173    // Why the fastest: noise (other programs, CPU frequency changes) only ever
174    // makes a run SLOWER, so the minimum is the cleanest measurement.
175    //
176    // The parameter type `&dyn Fn() -> f64` means "a reference to any function
177    // or closure that takes no arguments and returns an f64". `dyn` = the
178    // concrete type is decided at run time; we just need something callable.
179    //
180    // The body is an ITERATOR chain, read left to right:
181    //   (0..3)                  the numbers 0, 1, 2 (a RANGE: end excluded)
182    //   .map(|_| f())           for each number, ignore it (`_`) and call f()
183    //                           -> three timings, e.g. 812.0, 790.5, 801.2
184    //   .fold(f64::MAX, f64::min)
185    //                           combine them into one value: start from the
186    //                           biggest possible f64, then for each timing do
187    //                           `acc = f64::min(acc, timing)`:
188    //                             MAX   -> min(MAX, 812.0)   = 812.0
189    //                             812.0 -> min(812.0, 790.5) = 790.5
190    //                             790.5 -> min(790.5, 801.2) = 790.5  <- result
191    //                           (`f64::min` is passed as a function value,
192    //                           without calling it; fold calls it for us.)
193    // Iterators are lazy: nothing runs until `fold` pulls the values through.
194    //
195    // Why not simply `.min()`? It does not compile for f64. `.min()` requires
196    // `Ord` (a TOTAL order: any two values compare), and floats have NaN, which
197    // is neither less than, equal to, nor greater than anything, itself
198    // included. So f64 is only `PartialOrd`, and Rust makes you choose a NaN
199    // policy explicitly. `f64::min` ignores NaN. Alternatives:
200    //   .reduce(f64::min).unwrap()        // no seed; None if the list is empty
201    //   .min_by(f64::total_cmp).unwrap()  // IEEE total order: NaN sorts last
202    //
203    // Cost: zero. Range, map, fold and the closure are all inlined in a release
204    // build into the same loop you'd write by hand:
205    //   let mut m = f64::MAX; for _ in 0..3 { m = m.min(f()); }
206    // The one indirection is `&dyn Fn`: `f()` is a call through a function
207    // pointer (~1 ns, irrelevant next to multi-second runs). A generic
208    // `fn best<F: Fn() -> f64>(f: F)` would remove it, but closures cannot be
209    // generic, hence `dyn` here.
210    let best = |f: &dyn Fn() -> f64| (0..3).map(|_| f()).fold(f64::MAX, f64::min);
211
212    // ---------------------------------------------------------------------
213    // Step 3: time both layouts, two threads each.
214    // ---------------------------------------------------------------------
215    //
216    // This packs three closures inside each other; unpacked from the inside:
217    //
218    //   std::thread::scope(|s| { s.spawn(..); s.spawn(..); })
219    //       Starts two threads and WAITS for both to finish before returning.
220    //       `s` is the scope handle; `s.spawn(|| hammer(&adj.a))` starts a
221    //       thread that runs `hammer` on counter `a`, the other on `b`.
222    //       Why `scope`? The threads borrow `adj`, a local variable of `main`.
223    //       A plain `std::thread::spawn` thread could outlive `main`'s stack
224    //       frame, so Rust would reject the borrow. `scope` guarantees the
225    //       threads are joined before `adj` can go away, so borrowing is safe.
226    //
227    //   time(|| ...)
228    //       Our helper above: runs the closure and returns how long it took,
229    //       in milliseconds.
230    //
231    //   best(&|| ...)
232    //       Runs the timing 3 times and keeps the fastest. `&` because `best`
233    //       takes a *reference* to a closure (`&dyn Fn`).
234    //
235    // `||` with nothing between the pipes is a closure with no parameters.
236    // All these closures capture `adj`/`pad` BY REFERENCE (they borrow them),
237    // which is why no `move` keyword is needed.
238    let t_adj = best(&||
239        time(||
240            std::thread::scope(|s| {
241                s.spawn(|| hammer(&adj.a));
242                s.spawn(|| hammer(&adj.b));
243            })
244        )
245    );
246    let t_pad = best(&||
247        time(||
248            std::thread::scope(|s| {
249                s.spawn(|| hammer(&pad.a));
250                s.spawn(|| hammer(&pad.b));
251            })
252        )
253    );
254
255    // Both threads run at the same time, each doing ITERS increments, so:
256    //   ns/op  = latency of ONE increment as seen by one thread
257    //          = wall time / ITERS            (ms -> ns is * 1e6)
258    //   ops/ns = throughput of BOTH threads together
259    //          = (2 * ITERS) / wall time in ns
260    // Padded, each increment is an uncontended `lock add` (~3.5 ns on an i9-12900K);
261    // adjacent, each one also waits for the line to come back (~16 ns there).
262    let row = |label: &str, ms: f64| {
263        let ns = ms * 1e6;
264        println!("  {label:<28} {ms:8.1} {:8.2} {:8.3}", ns / ITERS as f64, 2.0 * ITERS as f64 / ns);
265    };
266    println!("2 threads x {ITERS} increments of their OWN counter:\n");
267    println!("  {:<28} {:>8} {:>8} {:>8}", "", "ms", "ns/op", "ops/ns");
268    row("adjacent (false sharing)", t_adj);
269    row("padded (separate lines)", t_pad);
270    println!("\n  slowdown from false sharing: {:.1}x", t_adj / t_pad);
271    println!("\n(Needs >= 2 physical cores; on 1 core there's nothing to bounce between.)");
272}
