1//! Message-passing litmus test: why acquire/release exist.
2//!
3//! A "litmus test" is a tiny concurrent program whose possible outcomes reveal
4//! what a memory model allows. This is the most important one:
5//!
6//! Writer:  data = 1;  flag = 1;
7//! Reader:  wait until flag == 1;  then read data.   Must it see data == 1?
8//!
9//! Intuition says "obviously yes: data was written first". But "first" only
10//! holds *inside the writer thread*. The CPU may make the two stores visible
11//! to other cores in a different order (e.g. `data` still sits in the
12//! writer's store buffer when `flag` reaches memory), and the compiler may
13//! swap two stores that look independent to it. The orderings are how you
14//! forbid that:
15//!
16//!   flag stored Release / loaded Acquire  -> YES, guaranteed by the language.
17//!   flag stored Relaxed / loaded Relaxed  -> NO guarantee. The reader may see
18//!                                            flag == 1 and stale data == 0.
19//!
20//! HONEST CAVEAT: x86-64 has Total Store Order — the hardware never reorders
21//! two stores or two loads — so on x86 you will almost certainly see 0
22//! violations even for Relaxed. The bug is still real: (a) the compiler may
23//! reorder Relaxed accesses, and (b) on ARM/POWER/RISC-V the CPU does. Run this
24//! on an AArch64 machine (e.g. an Arm cloud server) to see Relaxed fail; to check
25//! algorithms without such hardware use a model checker like `loom`.
26//! (The one reordering x86 does perform is a store followed by a load of a
27//! *different* address; that is what `SeqCst` fences exist for.)
28//!
29//! Run:  cargo run --release --bin litmus
30use std::sync::atomic::{AtomicU32, Ordering, Ordering::Relaxed};
31
32/// Number of independent message-passing rounds per configuration.
33const CELLS: usize = 4_000_000;
34
35/// Run the test with the given ordering for the flag's store and load.
36/// Returns how many times the reader saw the flag set but the data not yet.
37fn run(store: Ordering, load: Ordering) -> usize {
38    // One independent (data, flag) pair per iteration, so reader and writer
39    // chase each other through memory without any synchronisation but the flag.
40    // Everything is atomic, so even the "wrong" configuration has no undefined
41    // behaviour — it is allowed to give a stale answer, not to corrupt memory.
42    let data: Vec<AtomicU32> = (0..CELLS).map(|_| AtomicU32::new(0)).collect();
43    let flag: Vec<AtomicU32> = (0..CELLS).map(|_| AtomicU32::new(0)).collect();
44
45    // `thread::scope` lets the threads borrow `data`/`flag` from this function
46    // and joins them all before returning. The value of the scope is the value
47    // of its last expression: here, the reader's stale count.
48    std::thread::scope(|s| {
49        // Writer thread.
50        s.spawn(|| {
51            for i in 0..CELLS {
52                data[i].store(1, Relaxed); // (1) write payload
53                flag[i].store(1, store);   // (2) publish
54            }
55        });
56        // Reader thread; its return value is the stale count.
57        s.spawn(|| {
58            let mut stale = 0;
59            for i in 0..CELLS {
60                // Busy-wait for the writer to reach cell i. `spin_loop` is a
61                // hint to the CPU (x86 `pause`) that we are spinning; it saves
62                // power and helps a sibling hyper-thread.
63                while flag[i].load(load) == 0 { std::hint::spin_loop(); }
64                if data[i].load(Relaxed) == 0 { stale += 1; } // saw flag but not data!
65            }
66            stale
67        })
68        .join()
69        .unwrap()
70    })
71}
72
73fn main() {
74    println!("arch = {}", std::env::consts::ARCH);
75    println!("{CELLS} message-passing rounds per configuration\n");
76    let a = run(Ordering::Release, Ordering::Acquire);
77    println!("Release store / Acquire load : {a} stale reads   (must always be 0)");
78    let r = run(Relaxed, Relaxed);
79    println!("Relaxed store / Relaxed load : {r} stale reads   (allowed to be > 0; 0 is normal on x86)");
80}
