1//! Throughput: our SPSC ring vs std::sync::mpsc vs `Mutex<VecDeque>`.
2//!
3//! One producer thread sends 0..N; one consumer thread receives and checks the
4//! sum (0 + 1 + ... + N-1 = N(N-1)/2), which proves nothing was lost or
5//! duplicated.
6//!
7//! How each one waits when the queue is full/empty:
8//!   * SPSC ring and `Mutex<VecDeque>`: spin (busy-wait with a CPU pause hint).
9//!   * std::sync::mpsc::sync_channel: a real blocking channel — it spins
10//!     briefly, then parks the thread and is woken by the other side. That
11//!     wake-up machinery is part of what its number includes. (Since Rust 1.67
12//!     std's channel is based on crossbeam-channel, whose bounded flavour is a
13//!     lock-free array queue — which is why it comes so close to the ring.)
14//!
15//! Run:  cargo run --release --bin spsc_bench
16//! Caveat: numbers depend heavily on core placement (same core pair? SMT
17//! siblings? same CCX?). Treat ratios as indicative, not absolute.
18use lockfree::spsc;
19use std::collections::VecDeque;
20use std::sync::{mpsc, Mutex};
21use std::time::Instant;
22
23const N: u64 = 10_000_000;
24const CAP: usize = 1024; // every queue is bounded to this many items
25                         // Gauss's formula: 0 + 1 + ... + (N-1) = N(N-1)/2. Pair the first and last
26                         // terms (0 + (N-1)), the second and second-to-last (1 + (N-2)), ...: each pair
27                         // sums to N-1 and there are N/2 pairs. For N = 10^7 that's ~5 * 10^13, which
28                         // needs u64 (u32 tops out at ~4.3 * 10^9).
29                         // Comparing sums is a cheap check: a lost item would make the sum too small,
30                         // a duplicated one too large. (It would not catch reordering; the ring's
31                         // unit tests check strict FIFO order.)
32const EXPECT: u64 = N * (N - 1) / 2;
33
34/// Check the checksum and print time + throughput.
35/// Throughput = items / seconds / 10^6 ("M items/s"). `{name:<28}` pads the
36/// name to 28 characters, left-aligned; `{:>8.1}` right-aligns in 8 columns
37/// with 1 decimal place.
38fn report(name: &str, secs: f64, sum: u64) {
39    assert_eq!(sum, EXPECT, "{name}: lost or duplicated items!");
40    println!("{name:<28} {:>8.1} ms   {:>8.1} M items/s", secs * 1e3, N as f64 / secs / 1e6);
41}

43/// Our wait-free ring. `move` closures hand each thread its own end.
44fn bench_spsc() {
45    let (mut p, mut c) = spsc::channel::<u64>(CAP);
46    let t = Instant::now();
47    let sum = std::thread::scope(|s| {
48        s.spawn(move || {
49            for i in 0..N {
50                while p.push(i).is_err() {
51                    std::hint::spin_loop();
52                } // full: spin
53            }
54        });
55        s.spawn(move || {
56            let (mut got, mut sum) = (0, 0u64);
57            while got < N {
58                if let Some(v) = c.pop() {
59                    sum += v;
60                    got += 1;
61                } else {
62                    std::hint::spin_loop();
63                } // empty: spin
64            }
65            sum
66        })
67        .join()
68        .unwrap()
69    });
70    report("lock-free SPSC ring", t.elapsed().as_secs_f64(), sum);
71}

73/// The standard library's bounded channel (send blocks when full).
74fn bench_mpsc() {
75    let (tx, rx) = mpsc::sync_channel::<u64>(CAP);
76    let t = Instant::now();
77    let sum = std::thread::scope(|s| {
78        // When the producer finishes, `tx` is dropped, which ends `rx.iter()`.
79        s.spawn(move || {
80            for i in 0..N {
81                tx.send(i).unwrap();
82            }
83        });
84        s.spawn(move || rx.iter().sum::<u64>()).join().unwrap()
85    });
86    report("std::sync::mpsc (bounded)", t.elapsed().as_secs_f64(), sum);
87}

89/// The naive approach: one lock around a double-ended queue. Every push and
90/// pop contends on the same mutex (and the same cache line).
91///
92/// A bug that used to live here: the consumer spun while holding the lock.
93/// Its time jumped between ~1 s and 30+ s from run to run, even pinned to the
94/// same two P-cores. How it happened:
95///   * The guard from `if let Some(v) = q.lock().unwrap().pop_front()` lives
96///     until the end of the whole `if let ... else` (edition 2021), so the
97///     "queue empty → spin_loop()" branch ran with the lock held.
98///   * `std::sync::Mutex` is not fair: the thread that just unlocked usually
99///     wins the next lock too, because the lock's cache line is still in its
100///     core. So an empty-queue consumer could unlock and relock hundreds of
101///     millions of times while the producer rarely got in — and without the
102///     producer getting in, the queue stayed empty. A self-sustaining loop.
103///   * Whether a run fell into that loop depended on timing noise early on.
104///     Instrumented: fast runs had ~1–10 M empty polls; slow ones 60–300 M.
105///     Context switches were a few hundred either way — nobody slept; the
106///     time was all lock traffic.
107///
108/// Now the item is taken in its own statement, so the lock is released
109/// before spinning. The lessons: hold a lock only for the critical section,
110/// and watch where Rust drops temporaries. See the README section "Why the
111/// mutex time jumped around" for the measurements.
112fn bench_mutex() {
113    let q = Mutex::new(VecDeque::<u64>::with_capacity(CAP));
114    let t = Instant::now();
115    let sum = std::thread::scope(|s| {
116        s.spawn(|| {
117            for i in 0..N {
118                loop {
119                    let mut g = q.lock().unwrap();
120                    if g.len() < CAP {
121                        g.push_back(i);
122                        break;
123                    }
124                    drop(g); // release the lock before spinning, or the consumer can't drain
125                    std::hint::spin_loop();
126                }
127            }
128        });
129        s.spawn(|| {
130            let (mut got, mut tsum) = (0, 0u64);
131            while got < N {
132                // Take the item in its own statement so the lock is released at
133                // the `;`. Writing `if let Some(v) = q.lock().unwrap().pop_front()`
134                // instead would keep the guard (a temporary) alive until the end
135                // of the whole `if let ... else` in edition 2021 — so the else
136                // branch would spin *while holding the lock*. See bench_mutex's
137                // doc comment for what that did to the timings.
138                let item = q.lock().unwrap().pop_front();
139                if let Some(v) = item {
140                    tsum += v;
141                    got += 1;
142                } else {
143                    std::hint::spin_loop();
144                }
145            }
146            tsum
147        })
148        .join()
149        .unwrap()
150    });
151    report("Mutex<VecDeque> (spinning)", t.elapsed().as_secs_f64(), sum);
152}
153
154fn main() {
155    println!("{N} u64 items, ring capacity {CAP}\n");
156    bench_mutex();
157    bench_mpsc();
158    bench_spsc();
159}
