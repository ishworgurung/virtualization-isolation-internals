1//! Thin helpers over the cgroup v2 filesystem.
2//!
3//! A **cgroup** ("control group") is a set of processes that the kernel
4//! accounts for and limits *as a group*: total CPU time, memory, number of
5//! tasks, I/O bandwidth. Containers, systemd services and Firecracker microVMs
6//! are all confined this way. cgroups do not isolate what a process can *see*
7//! (that is namespaces' job); they limit what it can *consume*.
8//!
9//! The entire cgroup API is a file tree (normally `/sys/fs/cgroup`):
10//!   * `mkdir`  a directory        -> create a cgroup
11//!   * `rmdir`  it                 -> destroy it (must be empty of processes)
12//!   * write a PID to `cgroup.procs` -> move that process in
13//!   * write to `memory.max`, `cpu.max`, ... -> set limits
14//!   * read `memory.current`, `cpu.stat`, `*.pressure` ... -> observe
15//! No syscalls beyond open/read/write/mkdir/rmdir — which is why this crate has
16//! no dependencies.
17//!
18//! Children (threads and forked processes) start in their parent's cgroup, so
19//! placing one process in a group confines everything it later creates.
20

31/// Where the cgroup v2 filesystem ("cgroup2") is conventionally mounted.
32pub const CGROOT: &str = "/sys/fs/cgroup";
33
34/// Is this a cgroup **v2** (unified hierarchy) host?
35/// v2 has a single tree with a `cgroup.controllers` file at its root; v1 mounts
36/// one tree per controller (`/sys/fs/cgroup/cpu`, `/sys/fs/cgroup/memory`, ...).
37/// (A "hybrid" host mounts v1 trees there and v2 at `/sys/fs/cgroup/unified`;
38/// this function reports `false` for it.)
39pub fn is_v2() -> bool {
40    Path::new(CGROOT).join("cgroup.controllers").exists()
41}
42
43/// Absolute path of the cgroup this process currently lives in.
44/// `/proc/self/cgroup` on v2 has a single line `0::/some/path`
45/// (hierarchy id 0, empty controller list, path relative to the cgroup root).
46pub fn own_cgroup() -> io::Result<PathBuf> {
47    let s = fs::read_to_string("/proc/self/cgroup")?;
48    for line in s.lines() {
49        if let Some(rest) = line.strip_prefix("0::") {
50            return Ok(Path::new(CGROOT).join(rest.trim_start_matches('/')));
51        }
52    }
53    Err(io::Error::new(io::ErrorKind::NotFound, "no `0::` line in /proc/self/cgroup: this is not a pure cgroup v2 host"))
54}
55
56/// Are we running with effective UID 0?  The `Uid:` line of
57/// /proc/self/status lists real, effective, saved and filesystem UIDs;
58/// `nth(2)` after the "Uid:" label is the effective one.
59fn is_root() -> bool {
60    fs::read_to_string("/proc/self/status")
61        .ok()
62        .and_then(|s| s.lines().find(|l| l.starts_with("Uid:")).map(|l| l.split_whitespace().nth(2) == Some("0")))
63        .unwrap_or(false)
64}

66/// Find a cgroup subtree we are allowed to create children in.
67///
68/// systemd **delegates** `user@<uid>.service` to the user, so an unprivileged
69/// user owns that directory and may mkdir, set limits and move their own
70/// processes inside it. Override with `CGLAB_ROOT=<dir>`. Root can use `/sys/fs/cgroup`.
71/// (By default systemd enables only the `cpu`, `memory` and `pids` controllers
72/// for such a delegated subtree — which is why this lab has no `io.max` demo.)
73///
74/// Strategy: start from our own cgroup and walk up towards the root until a
75/// directory named `user@….service` is found.
76pub fn delegated_root() -> io::Result<PathBuf> {
77    if let Ok(p) = env::var("CGLAB_ROOT") {
78        return Ok(p.into());
79    }
80    let mut p = own_cgroup()?;
81    loop {
82        let name = p.file_name().and_then(|n| n.to_str()).unwrap_or("");
83        if name.starts_with("user@") && name.ends_with(".service") {
84            return Ok(p);
85        }
86        if !p.pop() || p == Path::new(CGROOT) {
87            break;
88        }
89    }
90    if is_root() {
91        return Ok(CGROOT.into());
92    }
93    Err(io::Error::new(
94        io::ErrorKind::PermissionDenied,
95        "no delegated cgroup subtree found; run as root or set CGLAB_ROOT to a cgroup directory you own",
96    ))
97}

99/// One cgroup directory.
100pub struct Group {
101    pub path: PathBuf,
102}
103
104impl Group {
105    /// `mkdir` = create a cgroup. The kernel populates the new directory with
106    /// its control files automatically; which ones appear depends on the
107    /// controllers the parent enabled in its `cgroup.subtree_control`.
108    pub fn create(parent: &Path, name: &str) -> io::Result<Group> {
109        let path = parent.join(name);
110        fs::create_dir(&path)?;
111        Ok(Group { path })
112    }
113
114    /// Write a control file, e.g. `set("memory.max", "64M")`.
115    /// Each write must be a single `write(2)`; `fs::write` of a short string is.
116    /// Invalid values are rejected by the kernel with an error (e.g. EINVAL),
117    /// which is returned here.
118    pub fn set(&self, file: &str, val: &str) -> io::Result<()> {
119        fs::write(self.path.join(file), val)
120    }
121
122    /// Read a control file, trimmed.
123    pub fn get(&self, file: &str) -> io::Result<String> {
124        Ok(fs::read_to_string(self.path.join(file))?.trim().to_string())
125    }
126
127    /// Parse a "flat keyed" file (`key value\n` per line) like `cpu.stat` or
128    /// `memory.events`. Non-numeric lines are skipped.
129    /// Missing files yield an empty map rather than an error.
130    pub fn kv(&self, file: &str) -> HashMap<String, u64> {
131        let mut m = HashMap::new();
132        if let Ok(s) = fs::read_to_string(self.path.join(file)) {
133            for line in s.lines() {
134                let mut it = line.split_whitespace();
135                if let (Some(k), Some(v)) = (it.next(), it.next()) {
136                    if let Ok(v) = v.parse() {
137                        m.insert(k.to_string(), v);
138                    }
139                }
140            }
141        }
142        m
143    }
144
145    /// Convenience: one value from a flat-keyed file (0 if absent).
146    pub fn stat(&self, file: &str, key: &str) -> u64 {
147        self.kv(file).get(key).copied().unwrap_or(0)
148    }
149
150    /// Kill everything in this cgroup and its descendants, then remove the
151    /// directories bottom-up. `cgroup.kill` (Linux 5.14+) atomically SIGKILLs the
152    /// whole subtree, avoiding the fork-race of walking `cgroup.procs`
153    /// (a process could fork a new child between us reading the list and
154    /// killing the entries).
155    ///
156    /// Child cgroups are subdirectories; control files are regular files, so
157    /// `is_dir()` tells them apart. Errors are ignored: this is best-effort
158    /// cleanup.
159    pub fn destroy(&self) {
160        if let Ok(rd) = fs::read_dir(&self.path) {
161            for e in rd.flatten() {
162                if e.path().is_dir() {
163                    Group { path: e.path() }.destroy();
164                }
165            }
166        }
167        let _ = fs::write(self.path.join("cgroup.kill"), "1");
168        // rmdir fails with EBUSY until the killed tasks have finished exiting
169        // (and until all child cgroups are gone). Retry for up to ~2 s.
170        for _ in 0..200 {
171            if fs::remove_dir(&self.path).is_ok() {
172                return;
173            }
174            sleep(Duration::from_millis(10));
175        }
176    }
177}

179/// A private cgroup subtree for one lab run: `<delegated>/cglab-<pid>`.
180///
181/// Note the shape (the v2 "no internal processes" rule in action):
182///
183/// ```text
184///   cglab-<pid>/          <- interior node: has controllers enabled, NO processes
185///     cpu-quota/          <- leaf: processes live here
186///     mem-high/           <- leaf
187/// ```
188///
189/// Removed automatically (with everything inside) when dropped.
190pub struct Sandbox {
191    pub group: Group,
192}
193
194impl Sandbox {
195    pub fn new() -> io::Result<Sandbox> {
196        let root = delegated_root()?;
197        // Make sure cpu/memory/pids are available to *our* children: a controller
198        // can only be enabled in a group if it is enabled in the parent's
199        // subtree_control. (Idempotent if it already is.)
200        let _ = fs::write(root.join("cgroup.subtree_control"), "+cpu +memory +pids");
201        let group = Group::create(&root, &format!("cglab-{}", std::process::id()))?;
202        // Enable the controllers for the leaves we will create below. This is
203        // what makes files like `cpu.max`/`memory.max` appear in the leaves.
204        group.set("cgroup.subtree_control", "+cpu +memory +pids").map_err(|e| {
205            io::Error::new(e.kind(), format!("enabling controllers in {}: {e}", group.path.display()))
206        })?;
207        Ok(Sandbox { group })
208    }
209
210    /// Create a leaf group under the sandbox.
211    pub fn leaf(&self, name: &str) -> io::Result<Group> {
212        Group::create(&self.group.path, name)
213    }
214}
215
216impl Drop for Sandbox {
217    fn drop(&mut self) {
218        self.group.destroy();
219    }
220}

222/// Spawn `argv` **already inside** `group`.
223///
224/// `pre_exec` runs in the forked child *after fork, before exec*. We write "0" to
225/// `cgroup.procs`, which means "move the calling process". So the program starts
226/// life confined and can never run a single instruction outside its limits.
227/// This is the same pattern Firecracker's `jailer` uses: it joins the cgroup
228/// first, then (after further lock-down) exec()s Firecracker.
229///
230/// The alternative — spawn first, then write the child's PID into
231/// `cgroup.procs` from the parent — leaves a window in which the child runs
232/// unconfined and may already have forked.
233pub fn spawn_in(group: &Group, argv: &[String]) -> io::Result<Child> {
234    let procs = group.path.join("cgroup.procs");
235    let mut cmd = Command::new(&argv[0]);
236    cmd.args(&argv[1..]);
237    // SAFETY: after fork() only the calling thread exists in the child, so code
238    // there must not depend on locks another thread might have held (e.g. the
239    // allocator's). The closure only opens a file and writes one byte (std
240    // converts short paths to C strings on the stack), and this lab spawns
241    // from a single-threaded parent, so no such lock can be held.
242    unsafe {
243        cmd.pre_exec(move || std::fs::write(&procs, "0"));
244    }
245    cmd.spawn()
246}

248/// Wait up to `d` for the child; `None` means it is still running.
249/// (std has no wait-with-timeout, so poll `try_wait` every 25 ms.)
250pub fn wait_timeout(child: &mut Child, d: Duration) -> io::Result<Option<ExitStatus>> {
251    let start = Instant::now();
252    loop {
253        if let Some(st) = child.try_wait()? {
254            return Ok(Some(st));
255        }
256        if start.elapsed() >= d {
257            return Ok(None);
258        }
259        sleep(Duration::from_millis(25));
260    }
