1//! cglab — a hands-on tour of cgroup v2.
2//!
3//! ```text
4//! cglab info                      what does this host/cgroup look like?
5//! cglab limits                    cpu.max, memory.high, memory.max (OOM), pids.max
6//! cglab weight                    cpu.weight: proportional sharing under contention
7//! cglab nip                       the "no internal processes" rule
8//! cglab gomaxprocs                what Go >= 1.25 (and Rust std) derive from cpu.max
9//! cglab jail --cpu 50 --mem 64 --pids 32 -- <cmd...>     jailer-style confinement
10//! cglab clean                     remove stale cglab-* groups
11//! cglab all                       info + limits + weight + nip + gomaxprocs
12//! ```
13//! Everything runs unprivileged inside a systemd-delegated subtree
14//! (`user@UID.service`), or anywhere you point `CGLAB_ROOT`.
15//!
16//! How the demos generate load: the binary re-runs *itself* with a hidden
17//! `worker` subcommand (see [`worker`]), started inside a freshly created
18//! cgroup via `cg::spawn_in`. The parent then reads the cgroup's statistics
19//! files to show what the kernel did to the worker.
20

34/// argv that re-invokes THIS binary as a worker, optionally behind a wrapper
35/// (e.g. `taskset -c 0`).
36/// Result: `[wrapper..., /path/to/cglab, args...]`.
37fn worker_argv(wrapper: &[&str], args: &[&str]) -> Vec<String> {
38    let exe = env::current_exe().unwrap().to_string_lossy().into_owned();
39    wrapper.iter().map(|s| s.to_string()).chain([exe]).chain(args.iter().map(|s| s.to_string())).collect()
40}

45    println!("\n\x1b[1m== {s} ==\x1b[0m");

48// ───────────────────────────────────────────────────────────── worker side ──
49// Hidden subcommands that run *inside* a cgroup and generate load.
50
51fn worker(args: &[String]) {
52    // n(i, default): numeric argument i, or `default` if missing/unparsable.
53    let n = |i: usize, d: u64| args.get(i).and_then(|s| s.parse().ok()).unwrap_or(d);
54    match args.first().map(|s| s.as_str()) {
55        // cpu <threads> <secs>: burn CPU on N threads.
56        Some("cpu") => {
57            let deadline = Instant::now() + Duration::from_secs(n(2, 2));
58            let hs: Vec<_> = (0..n(1, 1))
59                .map(|_| {
60                    std::thread::spawn(move || {
61                        let mut x = 1u64;
62                        while Instant::now() < deadline {
63                            // Pointless arithmetic (a linear congruential
64                            // generator step). black_box stops the optimiser
65                            // from deleting it, so the thread really computes.
66                            for _ in 0..10_000 {
67                                x = std::hint::black_box(x.wrapping_mul(6364136223846793005).wrapping_add(1));
68                            }
69                        }
70                        x
71                    })
72                })
73                .collect();
74            for h in hs {
75                let _ = h.join();
76            }
77        }
78        // mem <MiB> <secs>: allocate and keep touching memory.
79        Some("mem") => {
80            let (target, secs) = (n(1, 64), n(2, 3));
81            let start = Instant::now();
82            let deadline = start + Duration::from_secs(secs);
83            let mut chunks: Vec<Vec<u8>> = vec![];
84            while (chunks.len() as u64) < target && Instant::now() < deadline {
85                // A zeroed 1 MiB allocation is served by fresh mmap()ed pages
86                // that are not backed (or charged to the cgroup) until touched.
87                let mut c = vec![0u8; MIB as usize];
88                for i in (0..c.len()).step_by(4096) {
89                    c[i] = 1; // touch every page so it is really charged
90                }
91                chunks.push(c);
92                if chunks.len() % 8 == 0 {
93                    println!("    worker: {:>3} MiB resident   t={:>5} ms", chunks.len(), start.elapsed().as_millis());
94                }
95            }
96            // Keep the memory alive (and occasionally touched) until the deadline.
97            while Instant::now() < deadline {
98                for c in chunks.iter_mut() {
99                    std::hint::black_box(&mut c[0]);
100                }
101                sleep(Duration::from_millis(50));
102            }
103            println!("    worker: finished normally with {} MiB", chunks.len());
104        }
105        // pids <n> <secs>: try to start N threads.
106        // In Linux a thread is a task just like a process, and pids.max
107        // limits tasks, so thread creation fails (EAGAIN) at the limit.
108        Some("pids") => {
109            let (want, secs) = (n(1, 20), n(2, 1));
110            let mut ok = vec![];
111            let mut first_err = None;
112            for _ in 0..want {
113                match std::thread::Builder::new().spawn(move || sleep(Duration::from_secs(secs))) {
114                    Ok(h) => ok.push(h),
115                    Err(e) => {
116                        first_err.get_or_insert(e);
117                    }
118                }
119            }
120            println!("    worker: started {} of {} threads (+ main thread){}", ok.len(), want,
121                first_err.map(|e| format!("; first failure: {e}")).unwrap_or_default());
122            for h in ok {
123                let _ = h.join();
124            }
125        }
126        _ => eprintln!("unknown worker"),
127    }
128}

132/// `cglab info`: where we are in the hierarchy and what we are allowed to use.
133fn info() -> io::Result<()> {
134    header("host cgroup info");
135    println!("cgroup v2 unified hierarchy : {}", cg::is_v2());
136    if !cg::is_v2() {
137        println!("This host is cgroup v1 (or hybrid): one tree per controller under /sys/fs/cgroup/<controller>.");
138        return Ok(());
139    }
140    println!("this process's cgroup       : {}", cg::own_cgroup()?.display());
141    let root = cg::delegated_root()?;
142    println!("delegated root we can use   : {}", root.display());
143    let rg = Group { path: root.clone() };
144    // cgroup.controllers  = controllers this group MAY use (granted by its parent)
145    // cgroup.subtree_control = controllers it has turned on for its children
146    println!("controllers available there : {}", rg.get("cgroup.controllers").unwrap_or_default());
147    println!("controllers enabled below   : {}", rg.get("cgroup.subtree_control").unwrap_or_default());
148    println!("logical CPUs (std)          : {:?}  <- Rust std already honours cpu.max & affinity",
149        std::thread::available_parallelism());
150    // System-wide Pressure Stall Information. The first line ("some ...") is the
151    // share of time at least one task was stalled waiting for that resource.
152    for r in ["cpu", "memory", "io"] {
153        if let Ok(s) = fs::read_to_string(format!("/proc/pressure/{r}")) {
154            println!("system PSI {r:<6}: {}", s.lines().next().unwrap_or(""));
155        }
156    }
157    println!("(systemd delegated only the controllers listed above; `io` needs root or Delegate=io — so no io.max demo here.)");
158    Ok(())
159}

166/// `cglab limits`: one leaf cgroup per limit type, each with a worker that
167/// tries to exceed it.
168fn limits(sb: &Sandbox) -> io::Result<()> {
169    // ───────────────────────────────────────────────── cpu.max
170    header("cpu.max — hard CPU bandwidth cap (CFS quota / period)");
171    let g = sb.leaf("cpu-quota")?;
172    // Format: "<quota_us> <period_us>". 20000 100000 => at most 20 ms of CPU
173    // time per 100 ms window across ALL threads of the group = 0.2 CPUs.
174    // (Replaces v1's cpu.cfs_quota_us + cpu.cfs_period_us.)
175    // Once the group has used its quota, all its threads are "throttled"
176    // (not scheduled) until the next period starts.
177    g.set("cpu.max", "20000 100000")?;
178    println!("cpu.max = 20000 100000  (0.2 CPU).  Worker: 4 busy threads for 2 s of wall time.");
179    let t = Instant::now();
180    spawn_in(&g, &worker_argv(&[], &["worker", "cpu", "4", "2"]))?.wait()?;
181    let wall = t.elapsed().as_secs_f64();
182    // cpu.stat usage_usec = total CPU time consumed by the group, microseconds.
183    let used = g.stat("cpu.stat", "usage_usec") as f64 / 1e6;
184    println!("    wall {wall:.2} s, CPU consumed {used:.2} s  => average {:.2} CPUs (asked for 4, capped at 0.2)", used / wall);
185    // nr_periods: enforcement periods that elapsed while the group was runnable;
186    // nr_throttled: how many of those ended with the group throttled;
187    // throttled_usec: total time threads spent throttled (summed over threads).
188    println!("    cpu.stat: nr_periods={} nr_throttled={} throttled_usec={}",
189        g.stat("cpu.stat", "nr_periods"), g.stat("cpu.stat", "nr_throttled"), g.stat("cpu.stat", "throttled_usec"));
190    println!("    cpu.pressure  : {}", g.get("cpu.pressure").unwrap_or_else(|e| e.to_string()).replace('\n', "\n                    "));
191    println!("    (PSI: throttled/queued tasks show up as stall time — the 'noisy neighbour / starved tenant' signal.)");
192
193    // ───────────────────────────────────────────────── memory.high
194    header("memory.high — soft limit: THROTTLE + reclaim, never kill");
195    let g = sb.leaf("mem-high")?;
196    g.set("memory.high", &(32 * MIB).to_string())?;
197    let _ = g.set("memory.swap.max", "0"); // no swap: anon memory can't be reclaimed
198    println!("memory.high = 32 MiB. Worker wants 64 MiB. Watch it slow down once it crosses 32.");
199    let mut c = spawn_in(&g, &worker_argv(&[], &["worker", "mem", "64", "4"]))?;
200    // Sample memory.current WHILE the worker is alive (it reads 0 after the group empties).
201    let mut cur_mib = 0;
202    match wait_timeout(&mut c, Duration::from_secs(5))? {
203        Some(st) => println!("    worker exited: {st}"),
204        None => {
205            cur_mib = g.get("memory.current").ok().and_then(|s| s.parse::<u64>().ok()).unwrap_or(0) / MIB;
206            println!("    worker made no further progress for 5 s -> killing it (it was throttled, NOT OOM-killed)");
207            let _ = c.kill();
208            let _ = c.wait();
209        }
210    }
211    println!("    memory.current while stalled: {cur_mib} MiB (held at the high mark)");
212    print_events(&g, "memory.events");
213    println!("    ^ `high` counts reclaim/throttle episodes. `oom_kill` stays 0: memory.high never kills.");
214    println!("    Why it crawls: swap is off, and anonymous pages can't be reclaimed without swap, so the kernel");
215    println!("    can only *delay* the allocating task (up to 2 s per episode). With swap, or a file-backed");
216    println!("    working set, reclaim would make progress and the slowdown would be far gentler.");
217    println!("    memory.pressure: {}", g.get("memory.pressure").unwrap_or_default().replace('\n', "\n                     "));
218
219    // ───────────────────────────────────────────────── memory.max
220    header("memory.max — hard limit: reclaim, then the OOM killer");
221    let g = sb.leaf("mem-max")?;
222    g.set("memory.max", &(32 * MIB).to_string())?;
223    let _ = g.set("memory.swap.max", "0");
224    println!("memory.max = 32 MiB. Worker wants 128 MiB.");
225    let mut c = spawn_in(&g, &worker_argv(&[], &["worker", "mem", "128", "10"]))?;
226    let st = c.wait()?;
227    // signal() is Some(n) if the process was killed by signal n; 9 = SIGKILL.
228    match st.signal() {
229        Some(9) => println!("    worker died from SIGKILL (the cgroup OOM killer)"),
230        other => println!("    worker exit: {st} (signal {other:?})"),
231    }
232    // memory.events: `max` = times usage hit memory.max, `oom` = times the
233    // OOM path ran, `oom_kill` = processes killed by it.
234    print_events(&g, "memory.events");
235    // memory.peak (Linux 5.19+): highest usage ever recorded for this group.
236    println!("    memory.peak: {} MiB", g.get("memory.peak").ok().and_then(|s| s.parse::<u64>().ok()).unwrap_or(0) / MIB);
237    println!("    ^ the OOM kill is *contained*: it only considers tasks in this cgroup, so a tenant can't OOM its neighbours or the host.");
238    println!("    (check `dmesg` for 'Memory cgroup out of memory' if you have permission)");
239
240    // ───────────────────────────────────────────────── pids.max
241    header("pids.max — fork-bomb protection");
242    let g = sb.leaf("pids")?;
243    g.set("pids.max", "8")?;
244    println!("pids.max = 8 (threads count too). Worker tries to start 20 threads.");
245    spawn_in(&g, &worker_argv(&[], &["worker", "pids", "20", "1"]))?.wait()?;
246    print_events(&g, "pids.events");
247    println!("    ^ `max` counts fork/clone attempts rejected by the limit.");
248    Ok(())
249}

251/// `cglab weight`: proportional CPU sharing between two sibling groups.
252fn weight(sb: &Sandbox) -> io::Result<()> {
253    header("cpu.weight — proportional share, only matters under CONTENTION");
254    // Weights are *relative*: they only bite when CPUs are oversubscribed. On a
255    // 24-CPU box two single-threaded jobs never contend, so we force contention by
256    // pinning both to CPU 0 with `taskset`.
257    // (taskset sets the CPU affinity and then exec()s the worker, so the worker
258    // is still the process that spawn_in placed in the cgroup.)
259    let a = sb.leaf("w100")?;
260    let b = sb.leaf("w400")?;
261    a.set("cpu.weight", "100")?; // default weight is 100 (range 1..10000); replaces v1 cpu.shares
262    b.set("cpu.weight", "400")?;
263    println!("two groups, weights 100 vs 400, each one busy thread pinned to CPU 0 for 3 s");
264    let ca = spawn_in(&a, &worker_argv(&["taskset", "-c", "0"], &["worker", "cpu", "1", "3"]));
265    let cb = spawn_in(&b, &worker_argv(&["taskset", "-c", "0"], &["worker", "cpu", "1", "3"]));
266    let (mut ca, mut cb) = match (ca, cb) {
267        (Ok(x), Ok(y)) => (x, y),
268        _ => {
269            println!("  `taskset` not available; skipping.");
270            return Ok(());
271        }
272    };
273    ca.wait()?;
274    cb.wait()?;
275    // With 100:400 the scheduler gives the groups 1/5 and 4/5 of CPU 0.
276    let (ua, ub) = (a.stat("cpu.stat", "usage_usec") as f64, b.stat("cpu.stat", "usage_usec") as f64);
277    println!("    w100 used {:.2} s   w400 used {:.2} s   => ratio 1 : {:.1}  (expected ~1 : 4)", ua / 1e6, ub / 1e6, ub / ua);
278    Ok(())
279}

281/// `cglab nip`: demonstrate both halves of the no-internal-process rule.
282fn nip(sb: &Sandbox) -> io::Result<()> {
283    header("the v2 'no internal processes' rule");
284    println!("Rule: a cgroup may EITHER hold processes OR delegate controllers to children — not both.\n");
285
286    // Format an error with its raw errno (EBUSY = 16 is expected here).
287    let full = |e: &io::Error| format!("{} (errno {})", e, e.raw_os_error().unwrap_or(0));
288
289    // Case 1: group already has a process; try to enable controllers for children.
290    let g1 = sb.leaf("has-proc")?;
291    let mut sleeper = spawn_in(&g1, &["sleep".into(), "5".into()])?;
292    println!("1) `has-proc` contains a process (pid {}). Enable +memory for its children:", sleeper.id());
293    match g1.set("cgroup.subtree_control", "+memory") {
294        Ok(()) => println!("   unexpectedly succeeded"),
295        Err(e) => println!("   write to cgroup.subtree_control -> {}   <== refused", full(&e)),
296    }
297    let _ = sleeper.kill();
298    let _ = sleeper.wait();
299
300    // Case 2: enable controllers first, then try to put a process in.
301    let g2 = sb.leaf("delegating")?;
302    g2.set("cgroup.subtree_control", "+memory")?;
303    let leaf = Group::create(&g2.path, "leaf")?;
304    println!("\n2) `delegating` has +memory enabled for children. Try to move a process INTO it:");
305    // An ordinary child process (it starts in OUR cgroup); we then try to
306    // migrate it by writing its PID into the target's cgroup.procs.
307    let mut s2 = std::process::Command::new("sleep");
308    s2.arg("5");
309    let mut s2 = s2.spawn()?;
310    match g2.set("cgroup.procs", &s2.id().to_string()) {
311        Ok(()) => println!("   unexpectedly succeeded"),
312        Err(e) => println!("   write pid to delegating/cgroup.procs -> {}   <== refused", full(&e)),
313    }
314    match leaf.set("cgroup.procs", &s2.id().to_string()) {
315        Ok(()) => println!("   write pid to delegating/leaf/cgroup.procs   -> ok   (leaves hold processes)"),
316        Err(e) => println!("   leaf move failed: {}", full(&e)),
317    }
318    let _ = s2.kill();
319    let _ = s2.wait();
320
321    println!("\nWhy it exists: with controllers enabled, a process sitting in an interior group would\n\
322              compete for CPU/memory *against its own child groups* — v1's inconsistent semantics.\n\
323              The rule keeps accounting well-defined: resources are only ever consumed at leaves.\n\
324              (Exceptions: the root cgroup, and threaded subtrees for threaded controllers such as cpu and pids.)");
325    Ok(())
326}

328/// Effective CPU limit implied by `cpu.max` of this cgroup AND all ancestors
329/// (the tightest wins), in CPUs (quota / period). `None` = unlimited.
330/// This is the input Go >= 1.25 starts from when choosing its default
331/// GOMAXPROCS (Go then applies its own rounding and minimum).
332fn effective_cpu_limit(start: &std::path::Path) -> Option<f64> {
333    let mut best: Option<f64> = None;
334    let mut p = start.to_path_buf();
335    loop {
336        if let Ok(s) = fs::read_to_string(p.join("cpu.max")) {
337            let mut it = s.split_whitespace();
338            if let (Some(q), Some(per)) = (it.next(), it.next()) {
339                if q != "max" {
340                    if let (Ok(q), Ok(per)) = (q.parse::<f64>(), per.parse::<f64>()) {
341                        let lim = q / per;
342                        best = Some(best.map_or(lim, |b: f64| b.min(lim)));
343                    }
344                }
345            }
346        }
347        if p == std::path::Path::new(cg::CGROOT) || !p.pop() {
348            break;
349        }
350    }
351    best
352}

354/// `cglab gomaxprocs`: show how a CPU quota translates into "how many
355/// threads should my runtime run in parallel".
356fn gomaxprocs(sb: &Sandbox) -> io::Result<()> {
357    header("cpu.max -> GOMAXPROCS (the container-CPU footgun)");
358    println!("Before Go 1.25, GOMAXPROCS defaulted to the number of *host* CPUs even inside a\n\
359              container limited to, say, 2 CPUs: the runtime ran 24 Ps, the cgroup throttled\n\
360              them, and you got latency spikes. Go 1.25 defaults it to roughly min(NumCPU, ceil(cpu.max quota/period)).\n");
361    let ncpu = std::thread::available_parallelism().map(|n| n.get()).unwrap_or(1);
362    for (name, max) in [("half-cpu", "50000 100000"), ("two-and-a-half", "250000 100000"), ("unlimited", "max 100000")] {
363        let g = sb.leaf(name)?;
364        g.set("cpu.max", max)?;
365        let lim = effective_cpu_limit(&g.path);
366        // Our simple rule: ceil(limit), clamped to [1, host CPUs]. Go's actual
367        // rule may additionally enforce a minimum (see the note printed below).
368        let derived = lim.map_or(ncpu, |l| (l.ceil() as usize).clamp(1, ncpu));
369        println!("  cpu.max = {max:<14} effective limit {:>5}   -> derived parallelism {derived:>2}   (host has {ncpu})",
370            lim.map_or("none".to_string(), |l| format!("{l:.2}")));
371    }
372    println!("\nRust's std::thread::available_parallelism() also takes the cgroup CPU quota into account. Also");
373    println!("note: cpu.max is a *bandwidth* cap, not a CPU count — 4 threads × 0.5 CPU each still");
374    println!("get throttled after 50% of a core per period, so latency-sensitive services often");
375    println!("prefer requests (cpu.weight) without a hard limit. (I recall Go applying a floor of");
376    println!("2 when deriving this; check the Go 1.25 release notes for the exact rule.)");
377    Ok(())
378}

380/// `cglab jail --cpu PCT --mem MIB --pids N -- cmd args...`
381/// Run any command confined, the way Firecracker's jailer confines a VMM.
382/// Options come in `--flag value` pairs before the `--` separator.
383fn jail(args: &[String]) -> io::Result<()> {
384    let sep = args.iter().position(|a| a == "--").ok_or_else(|| io::Error::other("usage: cglab jail [--cpu PCT] [--mem MIB] [--pids N] -- cmd args..."))?;
385    let (opts, cmd) = (&args[..sep], &args[sep + 1..]);
386    if cmd.is_empty() {
387        return Err(io::Error::other("no command given"));
388    }
389    let sb = Sandbox::new()?;
390    let g = sb.leaf("jail")?;
391    let mut i = 0;
392    while i + 1 < opts.len() {
393        let v = &opts[i + 1];
394        match opts[i].as_str() {
395            // PCT% of one CPU = PCT*1000 us of CPU time per 100000 us period.
396            "--cpu" => g.set("cpu.max", &format!("{} 100000", v.parse::<u64>().unwrap_or(100) * 1000))?, // PCT of one CPU
397            "--mem" => {
398                g.set("memory.max", &(v.parse::<u64>().unwrap_or(64) * MIB).to_string())?;
399                let _ = g.set("memory.swap.max", "0"); // otherwise it could just swap
400            }
401            "--pids" => g.set("pids.max", v)?,
402            o => return Err(io::Error::other(format!("unknown option {o}"))),
403        }
404        i += 2;
405    }
406    println!("jail: {} (limits: cpu.max={:?} memory.max={:?} pids.max={:?})", g.path.display(),
407        g.get("cpu.max").unwrap_or_default(), g.get("memory.max").unwrap_or_default(), g.get("pids.max").unwrap_or_default());
408    let st = spawn_in(&g, cmd)?.wait()?;
409    println!("exit: {st}");
410    println!("usage_usec={} nr_throttled={} memory.peak={} MiB",
411        g.stat("cpu.stat", "usage_usec"), g.stat("cpu.stat", "nr_throttled"),
412        g.get("memory.peak").ok().and_then(|s| s.parse::<u64>().ok()).unwrap_or(0) / MIB);
413    print_events(&g, "memory.events");
414    print_events(&g, "pids.events");
415    Ok(())
416    // `sb` is dropped here, which kills anything left and removes the cgroups.
417}

419/// `cglab clean`: remove sandboxes left behind by an interrupted run (Ctrl-C
420/// skips the Drop-based cleanup).
421fn clean() -> io::Result<()> {
422    let root = cg::delegated_root()?;
423    for e in fs::read_dir(&root)?.flatten() {
424        if e.file_name().to_string_lossy().starts_with("cglab-") {
425            println!("removing {}", e.path().display());
426            Group { path: e.path() }.destroy();
427        }
428    }
429    Ok(())
430}

432/// Dispatch on the first argument (default: `all`).
433fn run() -> io::Result<()> {
434    let args: Vec<String> = env::args().skip(1).collect();
435    let cmd = args.first().map(|s| s.as_str()).unwrap_or("all");
436    if cmd == "worker" {
437        // We are a child process started by one of the demos, already inside
438        // its cgroup: just generate the requested load.
439        worker(&args[1..]);
440        return Ok(());
441    }
442    if cmd == "info" || cmd == "all" {
443        info()?;
444        if cmd == "info" {
445            return Ok(());
446        }
447    }
448    if cmd == "clean" {
449        return clean();
450    }
451    if cmd == "jail" {
452        return jail(&args[1..]);
453    }
454    if !cg::is_v2() {
455        return Err(io::Error::other("cgroup v2 required"));
456    }
457    let sb = Sandbox::new()?; // removed automatically on drop
458    println!("sandbox: {}", sb.group.path.display());
459    match cmd {
460        "limits" => limits(&sb)?,
461        "weight" => weight(&sb)?,
462        "nip" => nip(&sb)?,
463        "gomaxprocs" => gomaxprocs(&sb)?,
464        "all" => {
465            limits(&sb)?;
466            weight(&sb)?;
467            nip(&sb)?;
468            gomaxprocs(&sb)?;
469        }
470        _ => return Err(io::Error::other(format!("unknown command `{cmd}`; see the header of src/main.rs"))),
471    }
472    Ok(())
473}
474
475/// Entry point: print any error and exit non-zero.
476fn main() -> ExitCode {
477    match run() {
478        Ok(()) => ExitCode::SUCCESS,
479        Err(e) => {
480            eprintln!("cglab: {e}");
481            ExitCode::FAILURE
482        }
483    }
484}
